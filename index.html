// ========================================================
// 1. BUBBLE SORT ALGORITHM (อัลกอริทึมจัดเรียงแบบฟองสบู่)
// หลักการ: วนลูปเปรียบเทียบข้อมูล "คู่ที่อยู่ติดกัน" (Adjacent)
// หากตัวหน้ามากกว่าตัวหลัง จะทำการ Swap สลับตำแหน่ง
// ค่าที่มากที่สุดจะค่อย ๆ ลอยขึ้นไปอยู่ท้ายอาร์เรย์ในแต่ละรอบ
// ========================================================

function bubbleSort(arr) {
  const n = arr.length;
  let swapped;

  // Outer Loop: ควบคุมจำนวนรอบ (Pass) ทั้งหมด n-1 รอบ
  for (let i = 0; i < n - 1; i++) {
    swapped = false;

    // Inner Loop: เปรียบเทียบทีละคู่จากต้นแถวจนถึงขอบเขตที่ยังไม่จัดเรียง
    for (let j = 0; j < n - i - 1; j++) {
      
      // เปรียบเทียบ: หากตัวหน้า (arr[j]) มีค่ามากกว่าตัวหลัง (arr[j+1])
      if (arr[j].value > arr[j + 1].value) {
        
        // ทำการสลับตำแหน่ง (Swap)
        const temp = arr[j];
        arr[j] = arr[j + 1];
        arr[j + 1] = temp;

        swapped = true; // ทำเครื่องหมายว่ามีการสลับเกิดขึ้น
      }
    }

    // การเพิ่มประสิทธิภาพ: หากรอบนี้ไม่มีการ Swap เลย แปลว่าเรียงเสร็จสมบูรณ์แล้ว
    if (!swapped) break;
  }
  return arr;
}
// ========================================================
// 2. SELECTION SORT ALGORITHM (อัลกอริทึมจัดเรียงแบบเลือก)
// หลักการ: แบ่งอาร์เรย์เป็น 2 ส่วน คือ "Sorted" และ "Unsorted"
// ในแต่ละรอบจะสแกนหาตัวที่มีค่าน้อยที่สุด (Minimum) ในส่วน Unsorted
// แล้วนำมาสลับ (Swap) กับตำแหน่งหัวแถวของส่วน Unsorted นั้น
// ========================================================

function selectionSort(arr) {
  const n = arr.length;

  // Outer Loop: เลื่อนขอบเขตของส่วนที่จัดเรียงแล้วไปทีละช่อง
  for (let i = 0; i < n - 1; i++) {
    
    // กำหนดสมมติฐานว่าตำแหน่งเริ่มต้น i มีค่าน้อยที่สุด
    let minIndex = i;

    // Inner Loop: สแกนหาตำแหน่งที่ค่าน้อยที่สุดจริง ๆ ในส่วนที่เหลือ (i+1 ถึง n-1)
    for (let j = i + 1; j < n; j++) {
      if (arr[j].value < arr[minIndex].value) {
        minIndex = j; // จดจำ Index ที่พบค่าน้อยกว่า
      }
    }

    // หากพบตัวที่น้อยกว่าตำแหน่ง i จริง ให้ทำการ Swap สลับเข้าหัวแถว
    if (minIndex !== i) {
      const temp = arr[i];
      arr[i] = arr[minIndex];
      arr[minIndex] = temp;
    }
  }
  return arr;
}

🔍 แนวคิดของเกม (Core Concept):
ผู้เล่นรับบทเป็นนักสืบที่ต้องจัดเรียงลำดับกระดานแฟ้มคดีที่คนร้ายสับเปลี่ยนไว้ โดยใช้อัลกอริทึมการจัดเรียง (Sorting Algorithm) เป็นกฎเกณฑ์ตายตัวในการขยับข้อมูล

📁 คดีที่ 1 (Bubble Sort):
ผู้เล่นสามารถสลับตำแหน่งการ์ดได้ เฉพาะคู่ที่อยู่ติดกันเท่านั้น โดยต้องเปรียบเทียบและสลับเพื่อดันหลักฐานที่มีเวลาเกิดขึ้นทีหลัง (ค่ามากสุด) ไปไว้ด้านขวาทีละรอบ

📁 คดีที่ 2 (Selection Sort):
กระดานจะแบ่งเป็นโซนจัดเรียงแล้วและยังไม่จัดเรียง ผู้เล่นต้องค้นหาการ์ดที่ มีค่าน้อยที่สุด ในโซนที่ยังไม่ตรวจ แล้วสลับเข้าสู่หัวแถวเพื่อขยายโซนที่เสร็จสิ้นทีละตำแหน่ง

- ชนะ (Case Closed): เรียงลำดับถูกต้องครบทุกใบ
- แพ้ (Cold Case): สลับผิดกฎของอัลกอริทึมครบ 3 ครั้ง พลังชีวิตนักสืบจะหมดลง
